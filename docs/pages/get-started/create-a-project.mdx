import React, { useState } from 'react';
import { StyleSheet, View, Text, TextInput, Button, FlatList, Alert } from 'react-native';

export default function App() {
  const [entry, setEntry] = useState('');
  const [entries, setEntries] = useState([]);

  const addEntry = () => {
    if (entry.trim() === '') {
      Alert.alert('Oops!', 'Please write something to thank Jesus for!');
      return;
    }
    setEntries([...entries, { id: Date.now().toString(), text: entry }]);
    setEntry('');
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>🙏 Gratitude Journal for Jesus</Text>
      
      <TextInput
        style={styles.input}
        placeholder="What are you thankful for today?"
        value={entry}
        onChangeText={(text) => setEntry(text)}
      />

      <Button title="Add Entry" onPress={addEntry} color="#2ecc71" />

      <FlatList
        data={entries}
        renderItem={({ item }) => (
          <View style={styles.entryItem}>
            <Text>• {item.text}</Text>
          </View>
        )}
        keyExtractor={(item) => item.id}
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    padding: 20,
    backgroundColor: '#f0f9ff',
  },
  title: {
    fontSize: 24,
    fontWeight: 'bold',
    marginBottom: 20,
    color: '#2c3e50',
    textAlign: 'center',
  },
  input: {
    borderWidth: 1,
    borderColor: '#3498db',
    padding: 10,
    marginBottom: 10,
    borderRadius: 5,
    backgroundColor: 'white',
  },
  entryItem: {
    padding: 10,
    marginVertical: 5,
    backgroundColor: '#e8f4f8',
    borderRadius: 5,
  },
});
